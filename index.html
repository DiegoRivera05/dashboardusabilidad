""" normalizar_2663.py ================== Normaliza reporte_profesores_2663.csv y lo integra en el dashboard. Ejecutar desde la carpeta donde están los archivos: python normalizar_2663.py Requiere Python 3.7+. No necesita librerías externas. Archivos necesarios (en la misma carpeta o ajustar rutas): - reporte_profesores_2663.csv (nueva base 2026-3) - data_v2.json (datos actuales del dashboard) - dashboard_usabilidad.html (dashboard actual) Salidas: - reporte_profesores_2663_normalizado.csv (CSV limpio) - data_v3.json (JSON actualizado con 2026-3) - dashboard_usabilidad_v3.html (dashboard con nuevo semestre) """ import csv import json import os import re from pathlib import Path # ── Rutas ────────────────────────────────────────────────────────────────────── BASE = Path(__file__).parent CSV_ENTRADA = BASE / "reporte_profesores_2663.csv" JSON_ENTRADA = BASE / "data_v2.json" HTML_ENTRADA = BASE / "dashboard_usabilidad.html" CSV_SALIDA = BASE / "reporte_profesores_2663_normalizado.csv" JSON_SALIDA = BASE / "data_v3.json" HTML_SALIDA = BASE / "dashboard_usabilidad_v3.html" PERIODO_NUEVO = "2026-3" # ── Mapeo de columnas ────────────────────────────────────────────────────────── COL_MAP = { "Codigo_Profesor": "codigo_profesor", "Curso": "curso", "Categoria": "categoria", "Profesor": "profesor", "Correo": "correo_profesor", "Primer_Ingreso": "primer_ingreso", "Ultimo_Ingreso": "ultimo_ingreso", "Numero_de_Clicks":"numero_de_clicks", "Activo": "estado_profesor", } # ── Helpers ──────────────────────────────────────────────────────────────────── def limpiar(valor): """Reemplaza valores nulos y 'No registrado' por cadena vacía.""" if valor is None: return "" v = valor.strip() if v.upper() in ("NULL", "NO REGISTRADO", "N/A", ""): return "" return v def tiene_primer_ingreso(primer_ingreso_str): """Devuelve True si hay fecha de primer ingreso válida.""" return bool(limpiar(primer_ingreso_str)) def extraer_periodo_del_curso(curso): """Extrae el período del nombre del curso (ej: 'Materia-01234-2663' → '2026-3').""" PERIODOS = { "2663": "2026-3", "2661": "2026-1", "2561": "2025-1", "2463": "2024-2", "2461": "2024-1", "2363": "2023-2", "2361": "2023-1", } m = re.search(r"-(\d{4})$", curso.strip()) if m: return PERIODOS.get(m.group(1), m.group(1)) return PERIODO_NUEVO # ══════════════════════════════════════════════════════════════════════════════ # 1. Leer y normalizar CSV # ══════════════════════════════════════════════════════════════════════════════ print(f"\n📂 Leyendo {CSV_ENTRADA.name} ...") filas = [] with open(CSV_ENTRADA, encoding="utf-8-sig") as f: reader = csv.DictReader(f) for row in reader: fila_limpia = {} for col_orig, col_nuevo in COL_MAP.items(): fila_limpia[col_nuevo] = limpiar(row.get(col_orig, "")) fila_limpia["periodo"] = extraer_periodo_del_curso(fila_limpia["curso"]) filas.append(fila_limpia) print(f" ✓ {len(filas)} registros leídos") # Filtrar filas sin profesor ni código (aulas sin asignar) filas_validas = [f for f in filas if f["codigo_profesor"] or f["profesor"]] filas_sin_prof = len(filas) - len(filas_validas) print(f" ✓ {filas_sin_prof} registros sin profesor filtrados") print(f" ✓ {len(filas_validas)} registros válidos") # ── Guardar CSV normalizado ──────────────────────────────────────────────────── columnas = list(COL_MAP.values()) + ["periodo"] with open(CSV_SALIDA, "w", newline="", encoding="utf-8-sig") as f: writer = csv.DictWriter(f, fieldnames=columnas) writer.writeheader() writer.writerows(filas_validas) print(f"\n✅ CSV normalizado guardado: {CSV_SALIDA.name}") # ══════════════════════════════════════════════════════════════════════════════ # 2. Leer JSON existente # ══════════════════════════════════════════════════════════════════════════════ print(f"\n📂 Leyendo {JSON_ENTRADA.name} ...") with open(JSON_ENTRADA, encoding="utf-8") as f: data = json.load(f) print(f" ✓ {len(data['profesores'])} profesores existentes") print(f" ✓ Períodos existentes: {data.get('periodos', [])}") # ══════════════════════════════════════════════════════════════════════════════ # 3. Integrar nuevos datos # ══════════════════════════════════════════════════════════════════════════════ print(f"\n🔄 Integrando período {PERIODO_NUEVO} ...") # Índice de profesores por código idx_prof = {} for p in data["profesores"]: codigo = str(p.get("codigo_profesor", "")).strip() if codigo: idx_prof[codigo] = p # Procesar cada fila del CSV nuevos_profesores = 0 nuevas_aulas = 0 cursos_duplicados = 0 for fila in filas_validas: codigo = str(fila["codigo_profesor"]).strip() if not codigo: continue # Buscar o crear profesor if codigo not in idx_prof: nuevo_prof = { "codigo_profesor": codigo, "nombre_profesor": fila["profesor"], "correo_profesor": fila["correo_profesor"], "cursos": [] } data["profesores"].append(nuevo_prof) idx_prof[codigo] = nuevo_prof nuevos_profesores += 1 else: # Actualizar nombre/correo si estaban vacíos prof = idx_prof[codigo] if not prof.get("nombre_profesor") and fila["profesor"]: prof["nombre_profesor"] = fila["profesor"] if not prof.get("correo_profesor") and fila["correo_profesor"]: prof["correo_profesor"] = fila["correo_profesor"] prof = idx_prof[codigo] periodo = fila["periodo"] # Verificar si este curso ya existe para este período curso_nombre = fila["curso"] ya_existe = any( c.get("curso") == curso_nombre and c.get("periodo") == periodo for c in prof["cursos"] ) if ya_existe: cursos_duplicados += 1 continue # Construir registro de curso try: clicks = int(fila["numero_de_clicks"]) if fila["numero_de_clicks"] else 0 except ValueError: clicks = 0 curso_obj = { "curso": curso_nombre, "categoria": fila["categoria"], "cat_nombre": data.get("cat_names", {}).get(fila["categoria"], fila["categoria"]), "periodo": periodo, "clicks": clicks, "primer_ingreso": tiene_primer_ingreso(fila["primer_ingreso"]), "primer_ingreso_fecha": fila["primer_ingreso"], "ultimo_ingreso": fila["ultimo_ingreso"], "activo": fila["estado_profesor"].lower() in ("sí", "si", "s", "yes", "1", "true"), } prof["cursos"].append(curso_obj) nuevas_aulas += 1 print(f" ✓ Nuevos profesores agregados: {nuevos_profesores}") print(f" ✓ Nuevas aulas integradas: {nuevas_aulas}") print(f" ✓ Registros duplicados omitidos: {cursos_duplicados}") # ── Actualizar lista de períodos ─────────────────────────────────────────────── if PERIODO_NUEVO not in data.get("periodos", []): data.setdefault("periodos", []).append(PERIODO_NUEVO) data["periodos"].sort(reverse=True) print(f" ✓ Período {PERIODO_NUEVO} agregado") # ── Recalcular stats globales ────────────────────────────────────────────────── total_clicks = sum( c["clicks"] for p in data["profesores"] for c in p["cursos"] ) total_aulas = sum(len(p["cursos"]) for p in data["profesores"]) total_profs = len(data["profesores"]) data["stats"] = { "total_profesores": total_profs, "total_aulas": total_aulas, "total_clicks": total_clicks, "ultima_actualizacion": "25 de junio de 2026", } print(f"\n📊 Estadísticas actualizadas:") print(f" Profesores: {total_profs}") print(f" Aulas: {total_aulas}") print(f" Clics: {total_clicks:,}") # ── Guardar data_v3.json ─────────────────────────────────────────────────────── with open(JSON_SALIDA, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, separators=(",", ":")) size_mb = JSON_SALIDA.stat().st_size / 1_048_576 print(f"\n✅ JSON actualizado guardado: {JSON_SALIDA.name} ({size_mb:.2f} MB)") # ══════════════════════════════════════════════════════════════════════════════ # 4. Actualizar el HTML del dashboard # ══════════════════════════════════════════════════════════════════════════════ print(f"\n📂 Actualizando {HTML_ENTRADA.name} ...") with open(HTML_ENTRADA, encoding="utf-8") as f: html = f.read() # Localizar y reemplazar el bloque JSON embebido patron = r'(const DATA\s*=\s*)([\s\S]*?)(;\s*//\s*END DATA|;\s*\n\s*//|;\s*\n\s*const\s+EXCLUIDAS|;\s*\n\s*const\s+CAT_NAMES|;\s*\n\s*function)' json_nuevo = json.dumps(data, ensure_ascii=False, separators=(",", ":")) def reemplazar_data(m): return m.group(1) + json_nuevo + ";" nuevo_html, n = re.subn(r'const DATA\s*=\s*\{.*?\};', lambda m: f'const DATA = {json_nuevo};', html, count=1, flags=re.DOTALL) if n == 0: # Intento alternativo: buscar la asignación del JSON más grande del archivo # Suele estar como: const DATA = {...}; con el JSON de 4MB idx_inicio = html.find("const DATA = {") if idx_inicio == -1: idx_inicio = html.find("const DATA={") if idx_inicio != -1: # Encontrar el cierre del objeto buscando el balance de llaves pos = idx_inicio while pos < len(html) and html[pos] != '{': pos += 1 profundidad = 0 in_string = False escape = False string_char = None idx_fin = pos for i in range(pos, len(html)): c = html[i] if escape: escape = False continue if c == '\\' and in_string: escape = True continue if not in_string: if c in ('"', "'"): in_string = True string_char = c elif c == '{': profundidad += 1 elif c == '}': profundidad -= 1 if profundidad == 0: idx_fin = i break else: if c == string_char: in_string = False if idx_fin > pos: # Encontrar el ';' después del cierre idx_semicolon = html.find(';', idx_fin) if idx_semicolon == -1: idx_semicolon = idx_fin + 1 nuevo_html = ( html[:idx_inicio] + f"const DATA = {json_nuevo}" + html[idx_semicolon:] ) n = 1 print(" ✓ JSON reemplazado (método alternativo)") if n == 0: print(" ⚠️ No se pudo localizar 'const DATA = {...}' en el HTML.") print(" Guarda data_v3.json y usa el script inject_data.py para inyectarlo manualmente.") # Guardar HTML sin cambios de datos pero con nota HTML_SALIDA.write_text(html, encoding="utf-8") else: print(" ✓ JSON reemplazado correctamente") with open(HTML_SALIDA, "w", encoding="utf-8") as f: f.write(nuevo_html) size_html = HTML_SALIDA.stat().st_size / 1_048_576 print(f"\n✅ Dashboard actualizado: {HTML_SALIDA.name} ({size_html:.2f} MB)") print("\n🎉 ¡Listo! Archivos generados:") print(f" 📄 {CSV_SALIDA.name}") print(f" 📦 {JSON_SALIDA.name}") if HTML_SALIDA.exists(): print(f" 🌐 {HTML_SALIDA.name}") print("\nAbre dashboard_usabilidad_v3.html en tu navegador para ver los resultados.")
